== should format OPT with a multi-field condition ==
(ti=(毛衣) and pd=[20100101 to 20250315]) OPT (ab=编织)

[expect]
(
        ti = (毛衣)
    and pd = [20100101 to 20250315]
)
OPT (ab = (编织))

== should format OPT with a single-field condition ==
ti=(毛衣) OPT (ab=(编织 or 针织 or 交织 or 混纺 or knit* or "sweater production"))

[expect]
ti = (毛衣)
OPT (ab = (编织 or 针织 or 交织 or 混纺 or knit* or "sweater production"))

== should add the required parentheses around both sides of OPT ==
ti=(毛衣) and pd=[20200101 to 20241231] OPT ab=编织

[expect]
(
        ti = (毛衣)
    and pd = [20200101 to 20241231]
)
OPT (ab = (编织))

[expect]
ti = (毛衣)
OPT (
        ab = (编织)
    or  ap = (某某公司)
)
